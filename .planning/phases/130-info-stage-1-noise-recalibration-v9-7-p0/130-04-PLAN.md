from web/lib/classifier/wilson.ts:15-23:
  export function wilsonCiLower(n: number, k: number, z = Z_95): number {
    if (n === 0) return 0;
    const phat = k / n;
    const z2 = z * z;
    const denom = 1 + z2 / n;
    const center = phat + z2 / (2 * n);
    const radius = z * Math.sqrt((phat * (1 - phat) + z2 / (4 * n)) / n);
    return Math.max(0, (center - radius) / denom);
  }
Upper arm = (center + radius) / denom, clamped to <= 1 via Math.min(1, ...).
